-- BASE DE DATOS EDULINK

-- CREAR BASE DE DATOS
CREATE DATABASE IF NOT EXISTS edulink_payments_db
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_unicode_ci;



-- CREAR TABLAS 
CREATE TABLE IF NOT EXISTS cards (
  id INT AUTO_INCREMENT PRIMARY KEY,
  pan_hash CHAR(64) NOT NULL UNIQUE,
  last4 VARCHAR(4) NOT NULL,
  exp_mm CHAR(2) NOT NULL,
  exp_yy CHAR(2) NOT NULL,
  balance_cents BIGINT NOT NULL DEFAULT 0,
  status ENUM('active','blocked') NOT NULL DEFAULT 'active',
  created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);



CREATE TABLE IF NOT EXISTS authorizations (
  id BIGINT AUTO_INCREMENT PRIMARY KEY,
  card_id INT NOT NULL,
  amount_cents BIGINT NOT NULL,
  auth_code VARCHAR(12) NOT NULL,
  timestamp TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (card_id) REFERENCES cards(id)
);

-- CREAR TABLAS INSCRIPCIONES Y CERTIFICADOS 
CREATE TABLE inscripciones (
    id INT AUTO_INCREMENT PRIMARY KEY,
    estudiante_id INT NOT NULL,
    curso_id INT NOT NULL,
    estado VARCHAR(20) DEFAULT 'completado'
);

CREATE TABLE certificados (
    id INT AUTO_INCREMENT PRIMARY KEY,
    inscripcion_id INT NOT NULL,
    codigo VARCHAR(50) NOT NULL UNIQUE,
    fecha_emision DATE NOT NULL,
    FOREIGN KEY (inscripcion_id) REFERENCES inscripciones(id)
);
